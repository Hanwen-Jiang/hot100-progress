## 代码

```java
class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        if (matrix == null || matrix.length == 0 || matrix[0] == null || matrix[0].length == 0) return false;
        int m = matrix.length, n = matrix[0].length, left = 0, right = 1;
        int[][] queue = new int[n*m][2];
        while (left != right) {
            int[] q = queue[left++];
            int v = matrix[q[0]][q[1]];
            if (v < target) {
                if ((!(queue[right-1][0] == q[0] && queue[right-1][1] == q[1]+1)) && q[1]+1 < n) queue[right++] = new int[] { q[0], q[1]+1 };
                if (q[0]+1 < m) queue[right++] = new int[] { q[0]+1, q[1] };
            } else if (v == target) return true;
        }
        return false;
    }
}
```
