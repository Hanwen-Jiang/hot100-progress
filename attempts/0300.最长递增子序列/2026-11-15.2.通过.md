## 代码

```python
class Solution:
    def lengthOfLIS(self, nums: list[int]) -> int:
        def bisect_left(nums, k):
            l, r = 0, len(nums)
            while l < r:
                m = (l+r) >> 1
                l, r = (l, m) if k <= nums[m] else (m+1, r)
            return l
        dev = []
        for x in nums:
            k = dev[-1] if len(dev) != 0 else -inf
            if x > k:
                dev.append(x)
            else:
                dev[bisect_left(dev, x)] = x
        return len(dev)
```
