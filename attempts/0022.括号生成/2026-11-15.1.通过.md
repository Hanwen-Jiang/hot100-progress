## 代码

```python
class Solution:
    def generateParenthesis(self, n: int) -> list[str]:
        path, res = '', []
        ops, uses = ['(', ')'], {'(':n, ')':0}
        def dfs():
            nonlocal path
            if len(path) == 2*n:
                res.append(path)
            for c in ops:
                if uses[c] > 0:
                    if c == '(':
                        uses[')'] += 1
                    uses[c] -= 1
                    path+=c
                    dfs()
                    path=path[:-1]
                    uses[c] += 1
                    if c == '(':
                        uses[')'] -= 1
        dfs()
        return res
```
