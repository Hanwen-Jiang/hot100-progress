class Solution {
    public int[][] merge(int[][] intervals) {
        List<int[]> res = new ArrayList<>();
        Map<Integer,Integer> map = new TreeMap<>();
        for(int[] interval:intervals) map.put(interval[0],interval[1]);
        int[] current = new int[] {Integer.MIN_VALUE+2, Integer.MIN_VALUE+1};
        map.forEach((k,v)->{
            if(k>current[1] || v<current[0]) {
                if(current[0] <= current[1]) {res.add(current.clone());}
                current[0] = k;
                current[1] = v;
            } else {
                current[1] = Math.max(current[1],v);
            }
        });
        res.add(current);
        return res.toArray(new int[res.size()][2]);
    }

    static void main() {
        Solution solution = new Solution();
        System.out.println(Arrays.deepToString(solution.merge(new int[][]{{4, 7}, {1, 4}})));
        System.out.println(Arrays.deepToString(solution.merge(new int[][]{{1, 3}, {2, 6}, {8, 10}, {15, 18}})));
        System.out.println(Arrays.deepToString(solution.merge(new int[][]{{1, 4}, {4, 5}})));
    }
}
