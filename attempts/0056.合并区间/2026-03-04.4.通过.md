```java
    class Solution {
        public int[][] merge(int[][] intervals) {
            if (intervals == null || intervals.length == 0) return new int[0][2];
            boolean isSorted = true;
            for (int i = 1; i < intervals.length; i++) {
                if (intervals[i][0] < intervals[i - 1][0]) {
                    isSorted = false;
                    break;
                }
            }
            if (!isSorted) {
                Arrays.sort(intervals, Comparator.comparingInt(a -> a[0]));
            }
            int[][] tempResult = new int[intervals.length][2];
            int resultIndex = 0;
            tempResult[resultIndex] = intervals[0];
            for (int i = 1; i < intervals.length; i++) {
                if (intervals[i][0] > tempResult[resultIndex][1]) {
                    resultIndex++;
                    tempResult[resultIndex] = intervals[i];
                } else {
                    tempResult[resultIndex][1] = Math.max(tempResult[resultIndex][1], intervals[i][1]);
                }
            }
            return Arrays.copyOf(tempResult, resultIndex + 1);
        }
    }
```
