class Solution {
    public ListNode mergeTwoLists(ListNode list1, ListNode list2) {
        ListNode head, p;
        if (list1 == null || list2 == null) {
            if (list1 == null) return list2;
            else return list1;
        }
        if (list1.val < list2.val) {
            p = head = list1;
            list1 = list1.next;
        } else {
            p = head = list2;
            list2 = list2.next;
        }
        while (list1 != null && list2 != null) {
            if (list1.val < list2.val) {
                p.next = list1;
                list1 = list1.next;
            } else  {
                p.next = list2;
                list2 = list2.next;
            }
            p = p.next;
        }
        return head;
    }
}
