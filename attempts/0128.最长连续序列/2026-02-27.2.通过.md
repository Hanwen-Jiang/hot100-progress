class Solution {
    class UnionFind {
        Map<Integer, Integer> parent = new HashMap<>();
        Map<Integer, Integer> size = new HashMap<>();
        int maxSubSize = 0;
        public void add(int x) {
            if (!parent.containsKey(x)) {
                parent.put(x, x);
                size.put(x, 1);
                maxSubSize = Math.max(maxSubSize, 1);
            }
        }
        public int find(int x) {
            if (parent.get(x) == x) return x;
            parent.put(x, find(parent.get(x)));
            return parent.get(x);
        }
        public void union(int x, int y) {
            int rootX = find(x);
            int rootY = find(y);
            if (rootX != rootY) {
                parent.put(rootX, rootY);
                int newSize = size.get(rootX) + size.get(rootY);
                size.put(rootY, newSize);
                maxSubSize = Math.max(maxSubSize, newSize);
            }
        }
    }

    public int longestConsecutive(int[] nums) {
        if (nums.length == 0) return 0;
        UnionFind uf = new UnionFind();
        for (int num : nums) uf.add(num);
        for (int num : nums) if (uf.parent.containsKey(num + 1)) uf.union(num, num + 1);
        return uf.maxSubSize;
    }
}
