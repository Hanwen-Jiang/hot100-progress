class Solution {
    public boolean isSymmetric(TreeNode root) {
        // 3 2 4 1 4 2 3
        // 2 3 1 2 3
        List<Integer> lefts = new ArrayList<>(), rights = new ArrayList<>();
        @FunctionalInterface
        interface InterFun {
            void run(TreeNode x);
        }

        InterFun[] fun = new InterFun[2];
        fun[0] = x -> {
            if (x == null) return;
            fun[0].run(x.left);
            lefts.add(x.val);
            fun[0].run(x.right);
        };
        fun[1] = x -> {
            if (x == null) return;
            fun[1].run(x.right);
            rights.add(x.val);
            fun[1].run(x.left);
        };
        fun[0].run(root.left);
        fun[1].run(root.right);
        return lefts.equals(rights);
    }
}
