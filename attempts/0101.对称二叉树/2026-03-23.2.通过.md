class Solution {
    public boolean isSymmetric(TreeNode root) {
        // 3 2 4 1 4 2 3
        // 2 3 1 2 3
        List<Integer> lefts = new ArrayList<>(), rights = new ArrayList<>();
        int depth = treeDeep(root);
        @FunctionalInterface
        interface InterFun {
            void run(TreeNode x, int curDeep);
        }

        InterFun[] fun = new InterFun[2];
        fun[0] = (x, d) -> {
            if (x == null) {
                if(d < depth)
                    for (int i = 0; i < (int) (Math.pow(2, depth-d)-1); i++) lefts.add(null);
                return;
            }
            fun[0].run(x.left, d+1);
            lefts.add(x.val);
            fun[0].run(x.right, d+1);
        };
        fun[1] = (x, d) -> {
            if (x == null) {
                if(d < depth)
                    for (int i = 0; i < (int) (Math.pow(2, depth-d)-1); i++) rights.add(null);
                return;
            }
            fun[1].run(x.right, d+1);
            rights.add(x.val);
            fun[1].run(x.left, d+1);
        };
        fun[0].run(root.left, 1);
        fun[1].run(root.right, 1);
        return lefts.equals(rights);
    }

    private int treeDeep(TreeNode root) {
        if (root==null) return 0;
        return Math.max(treeDeep(root.right), treeDeep(root.left)) + 1;
    }
}
