```java
class Solution {
    public List<List<String>> groupAnagrams(String[] strs) {
        HashMap<Integer, HashMap<Integer, List<String>>> map = new HashMap<>();
        for (String s  : strs) {
            int len = s.length(), charLen = 0;
            char[] chars = s.toCharArray();
//            Arrays.sort(chars);
            for (char c : chars) charLen+=c;
            HashMap<Integer, List<String>> temp = map.containsKey(len) ? map.get(len) : new HashMap<>();
            List<String> stringList = temp.containsKey(charLen) ? temp.get(charLen) : new ArrayList<>();
            stringList.add(s);
            temp.put(charLen, stringList);
            map.put(len, temp);
        }
        return map.values().stream()
                .flatMap(innerMap -> innerMap.values().stream())
                .toList();
    }
}
```
