class Solution {
    public int lengthOfLongestSubstring(String s) {
        HashSet<Character> sset = new HashSet<>();
        int len = 0, llen = 0;
        for(int i=0; i<s.length(); i++) {
            char c = s.charAt(i);
            if(sset.contains(c)) {
                sset.clear();
                sset.add(c);
                len = 1;
            } else {
                len++;
                sset.add(c);
            }
            llen = len > llen ? len : llen;
        }
        return llen;
    }
}
