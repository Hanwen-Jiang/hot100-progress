```java
class Solution {
    public boolean isValid(String s) {
        if (s.length()%2 == 1) return false;
        char[] cs = new char[s.length()/2];
        int idx = 0;
        for (char c: s.toCharArray()) {
            if (idx != cs.length && (c == '(' || c == '[' || c == '{')) cs[idx++] = c;
            else if (idx != 0 && ((c == ')' && cs[idx-1] == '(') || (c == ']' && cs[idx-1] == '[') || (c == '}' && cs[idx-1] == '{'))) idx--;
            else return false;
        }
        return idx == 0;
    }
}
```
