## 代码

```java
/*
// Definition for a Node.
class Node {
    int val;
    Node next;
    Node random;

    public Node(int val) {
        this.val = val;
        this.next = null;
        this.random = null;
    }
}
*/

class Solution {
    public Node copyRandomList(Node head) {
        if (head == null) return null;
        HashMap<Node, Node> map = new HashMap<Node, Node>();
        Node dummy = new Node(-1), p = head, p1 = dummy;
        while (p != null) {
            Node cur = new Node(p.val);
            cur.next = p.next;
            map.put(p, cur);
            p1.next = cur;
            p1 = cur;
            p = p.next;
        }
        p = head;p1 = dummy.next;
        while (p != null) {
            p1.random = map.getOrDefault(p.random, null);
            p1 = p1.next;
            p = p.next;
        }
        return dummy.next;
    }
}
```
