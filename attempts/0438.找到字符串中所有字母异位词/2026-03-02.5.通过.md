class Solution {
    public List<Integer> findAnagrams(String s, String p) {
    int sLen = s.length(), pLen = p.length();
    List<Integer> ans = new ArrayList<>();
    if (sLen < pLen) return ans;
    int[] map = new int[26];
    int[] map_ = new int[26]; 
    for (int i = 0; i < pLen; i++) map[p.charAt(i) - 'a']++;
    int pl = pLen; 
    int left = 0; 
    for (int i = 0; i < sLen; i++) {
        int c = s.charAt(i) - 'a';
        if (map[c] == 0) {
            if (pl != pLen) {
                java.util.Arrays.fill(map_, 0);
                pl = pLen;
            }
            left = i + 1; 
        } else {
            map_[c]++;
            pl--;
            while (map_[c] > map[c]) {
                map_[s.charAt(left) - 'a']--;
                pl++;
                left++;
            }
            if (pl == 0) {
                ans.add(left);
                map_[s.charAt(left) - 'a']--;
                pl++;
                left++;
            }
        }
    }
    return ans;
}
}
