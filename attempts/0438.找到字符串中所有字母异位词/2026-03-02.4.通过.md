class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        char[] ss = s.toCharArray(), ps = p.toCharArray();
        int[] map = new int[26];
        int[] map_ = new int[26];
        for (char c : ps) map[c - 'a']++;
        List<Integer> ans = new ArrayList<>();
        int pl = ps.length;
        int left = 0;
        for (int i = 0; i < ss.length; i++) {
            int c = ss[i] - 'a';
            if (map[c] == 0) {
                if (pl != ps.length) {
                    for (int j = left; j < i; j++) {
                        map_[ss[j] - 'a'] = 0;
                    }
                    pl = ps.length;
                }
                left = i + 1;
            } else {
                map_[c]++;
                pl--;
                while (map_[c] > map[c]) {
                    map_[ss[left] - 'a']--;
                    pl++;
                    left++;
                }
                if (pl == 0) {
                    ans.add(left);
                    map_[ss[left] - 'a']--;
                    pl++;
                    left++;
                }
            }
        }
        return ans;
    }
}
