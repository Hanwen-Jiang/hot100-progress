class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        char[] ss = s.toCharArray(),  ps = p.toCharArray();
        int pl = ps.length;
        boolean lastPong = false;
        int[] map = new int[26], map_ = new int[26];
        List<Integer> ans = new ArrayList<>();
        for (char c : ps) map[c - 'a']++;
        for (int i = 0; i < ss.length; i++) {
            int cur = map[ss[i] - 'a'];
            if (cur != 0) {
                //命中 abcba
                if ((map_[ss[i]-'a'] < cur && lastPong) || (!lastPong && pl == ps.length)) {
                    map_[ss[i]-'a']++;
                    lastPong = true;
                    pl--;
                } else continue;
                if (pl == 0) {
                    ans.add(i-ps.length+1);
                    map_[ss[i-ps.length+1]-'a']--;
                    pl = 1;
                }
            } else {
                //未命中
                lastPong = false;
                if (pl != ps.length) {
                    map_ = new int[26];
                    pl =  ps.length;
                }
            }
        }
        return ans;
    }

    static void main() {
        Solution solution = new Solution();
        System.out.println(solution.findAnagrams("abacbabc", "abc"));
        System.out.println(solution.findAnagrams("cbaebabacd", "abc"));
        System.out.println(solution.findAnagrams("abab", "ab"));
    }
}
