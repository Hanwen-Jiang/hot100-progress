class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        char[] ss = s.toCharArray(),  ps = p.toCharArray();
        int pl = ps.length, lastPongIndex = 0;
        boolean lastPong = false;
        int[] map = new int[26], map_ = new int[26];
        List<Integer> ans = new ArrayList<>();
        for (char c : ps) map[c - 'a']++;
        for (int i = 0; i < ss.length; i++) {
            int cur = map[ss[i] - 'a'];
            if (cur != 0) {
                //命中
                if ((map_[ss[i]-'a'] < cur && lastPong) || (!lastPong && pl == ps.length)) {
                    map_[ss[i]-'a']++;
                    lastPong = true;
                    pl--;
                } else {
                    int k = i-ps.length+1;
                    if (k>=lastPongIndex && map_[ss[k]-'a'] > 0 && ss[k] != ss[i]) {
                        map_[ss[k]-'a']--;
                        pl++;
                    }
                    continue;
                }
                if (pl == 0) {
                    ans.add(i-ps.length+1);
                    map_[ss[i-ps.length+1]-'a']--;
                    pl = 1;
                }
            } else {
                //未命中
                lastPong = false;
                lastPongIndex = i;
                if (pl != ps.length) {
                    map_ = new int[26];
                    pl =  ps.length;
                }
            }
        }
        return ans;
    }
}
