class Solution {
    public String longestPalindrome(String s) {
        HashMap<Character, Integer[]> map = new HashMap<>();
        int maxLen = -1;
        Integer[] interval = new Integer[2];
        AtomicReference<Character> result = new AtomicReference<>((char) 0);
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (map.containsKey(c)) {
                Integer[] integers = map.get(c);
                integers[1] = i;
                map.put(c, integers);
            }
            else map.put(c, new Integer[]{i, i});
        }
        for (Integer[] value : map.values()) {
            if (maxLen < value[1]-value[0]) {
                maxLen = value[1] - value[0];
                interval = value;
            }
        }
        return s.substring(interval[0], interval[1]+1);
    }
}
