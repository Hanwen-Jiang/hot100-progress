class Solution {
    public String longestPalindrome(String s) {
        int length = s.length(), len = 3;
        if (length == 0) return "";
        if (length == 1) return s;
        if (length == 2) {
            if (s.charAt(0) == s.charAt(1)) return s;
            else return s.substring(0, 1);
        }
        int[] interval = new int[]{0, 0};
        Queue<Integer> queue1 = new LinkedList<>();

        for (int i = 1; i < length-1; i++) queue1.add(i);
        Queue<Integer> queue2 = new LinkedList<>();

        for (int i = 0; i < length-1; i++) if(s.charAt(i) == s.charAt(i+1)) {
            queue2.add(i);
            interval =  new int[]{i, i+1};
        }
        for (len=3; len<=length; len++) {
            if (len%2 == 1) {
                Queue<Integer> queue12 = new LinkedList<>();
                while (!queue1.isEmpty()) {
                    Integer poll = queue1.poll();
                    // 3 -> -1 +1 5 -> -2 +2
                    if (s.charAt(poll - len/2) == s.charAt(poll + len/2)) {
                        if (poll - len/2-1 >= 0 && poll + len/2+1 < length) queue12.add(poll);
                        interval = new int[]{poll - len/2, poll + len/2};
                    }
                }
                queue1 = queue12;
            } else {
                Queue<Integer> queue22 = new LinkedList<>();
                while (!queue2.isEmpty()) {
                    Integer poll = queue2.poll();
                    if (s.charAt(poll - len/2 + 1) == s.charAt(poll + len/2)) {
                        if (poll - len/2 >= 0 && poll + len/2 + 1 < length) queue22.add(poll);
                        interval = new int[]{poll - len/2 + 1, poll + len/2};
                    }
                }
                queue2 = queue22;
            }
        }

        return s.substring(interval[0], interval[1] + 1);
    }
}
