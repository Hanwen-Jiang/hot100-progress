```java
class Solution {
    public String longestPalindrome(String s) {
        HashMap<Character, Integer> map = new HashMap<>();
        int[] interval = new int[]{0, 0};
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (map.containsKey(c)) {
                int thislen = i-map.get(c);
                if (thislen > interval[0]) {
                    interval = new int[]{thislen, map.get(c)};
                }
            }
            map.put(c, i);
        }
        return s.substring(interval[1], interval[0]+interval[1]+1);
    }
}
```
