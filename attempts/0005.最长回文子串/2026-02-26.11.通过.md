```java
class Solution {
    public String longestPalindrome(String s) {
        int n = s.length();
        if (n < 2) return s;
        char[] sChars = s.toCharArray();
        char[] t = new char[n * 2 + 3];
        t[0] = '^';
        t[1] = '#';
        for (int i = 0; i < n; ++i) {
            t[i * 2 + 2] = sChars[i];
            t[i * 2 + 3] = '#';
        }
        t[n * 2 + 2] = '$';

        int[] halfLen = new int[n * 2 + 1];
        int centerIndex = 0;
        int C = 0;
        int R = 0;

        for (int i = 2; i < halfLen.length; ++i) {
            int hl = 1;
            if (i < R) hl = Math.min(halfLen[C * 2 - i], R - i);
            while (t[i - hl] == t[i + hl]) hl++;
            if (i + hl > R) {
                C = i;
                R = i + hl;
            }
            halfLen[i] = hl;
            if (hl > halfLen[centerIndex]) centerIndex = i;
        }

        int maxHl = halfLen[centerIndex];
        return s.substring((centerIndex - maxHl) / 2, (centerIndex + maxHl) / 2 - 1);
    }
}
```
