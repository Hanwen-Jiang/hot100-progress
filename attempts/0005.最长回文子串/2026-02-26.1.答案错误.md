```java
class Solution {
    public String longestPalindrome(String s) {
        int length = s.length()-1, len = 1;
        int[] interval = new int[]{0, 0};
        boolean[][] dp = new boolean[length+1][length+1];
        for (int i = 0; i <= length; i++) dp[i][i] = true;
        for (int i = 0; i < length; i++) {
            if (s.charAt(i)==s.charAt(i+1)) {
                dp[i][i + 1] = true;
                len = 2;
                interval = new int[]{i, i+1};
            }
        }
        for (int i = 0; i < length-1; i++) {
            for (int j = i+2; j < length+1; j++) if (dp[i+1][j-1]) {
                if (s.charAt(i) == s.charAt(j)) {
                    dp[i][j] = true;
                    if (len < j - i + 1) {
                        len = j - i + 1;
                        interval = new int[]{i, j};
                    }
                }
            }
        }
        return s.substring(interval[0], interval[1]+1);
    }
}
```
