## 代码

class Solution {
    public String longestPalindrome(String s) {
        if (s == null || s.length() == 0) return "";
        int n = s.length();
        int virtualLen = 2 * n + 1;
        int[] p = new int[virtualLen];
        int C = 0, R = 0;
        int maxLen = 0, centerIndex = 0;

        for (int i = 0; i < virtualLen; i++) {
            int iMirror = 2 * C - i;
            p[i] = R > i ? Math.min(R - i, p[iMirror]) : 0;
            int left = i - 1 - p[i];
            int right = i + 1 + p[i];
            while (left >= 0 && right < virtualLen) {
                if (left % 2 == 0) p[i]++;
                else if (s.charAt(left / 2) == s.charAt(right / 2)) p[i]++;
                else break;
                left--;
                right++;
            }
            if (i + p[i] > R) {
                C = i;
                R = i + p[i];
            }
            if (p[i] > maxLen) {
                maxLen = p[i];
                centerIndex = i;
            }
        }

        int start = (centerIndex - maxLen) / 2;
        return s.substring(start, start + maxLen);
    }
}
