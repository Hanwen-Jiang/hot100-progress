    public int[] maxSlidingWindow(int[] nums, int k) {
        int n = nums.length;
        if (n == 0) return new int[0];
        if (k == 1) return nums.clone();
    
        int[] ans = new int[n - k + 1];
        int maxVal = Integer.MIN_VALUE, maxIdx = -1;
    
        for (int left = 0, right = k - 1; right < n; left++, right++) {
            if (maxIdx >= left) {                 // 旧最大值还在窗口
                if (nums[right] >= maxVal) {      // 新进来的更大/相等
                    maxVal = nums[right];
                    maxIdx = right;
                }
            } else {                               // 旧最大值出窗口，重扫
                maxVal = nums[left];
                maxIdx = left;
                for (int j = left + 1; j <= right; j++) {
                    if (nums[j] >= maxVal) {
                        maxVal = nums[j];
                        maxIdx = j;
                    }
                }
            }
            ans[left] = maxVal;
        }
        return ans;
    }
}
