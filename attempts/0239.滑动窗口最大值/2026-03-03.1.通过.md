## 代码

```java
class Solution {
    // [0, n-1] n-k+1 k <= [1, n-1] n-k+1 k-1 + [0, n-k+1] n-k+1 1
    // [1, n-1] n-k+1 k-1 <= [2, n-1] n-k+1 k-2 + [1, n-k+1] n-k+1 1
    // 3 -1 5 5 6 7
    // 1 3 -1 -3 5 3 6 7
    public int[] maxSlidingWindow(int[] nums, int k) {
        int len = nums.length, head = 0, tail = 0;
        int[] ans = new int[len - k + 1], dq = new int[len];
        for (int i = 0; i < len; i++) {
            if (head < tail && dq[head] < i-k+1) head++;
            while (head < tail && nums[dq[tail-1]] <= nums[i]) tail--;
            dq[tail++] = i;
            if (i>=k-1) ans[i-k+1] = nums[dq[head]];
        }
        return ans;
    }

    static void main() {
        Solution solution = new Solution();
        System.out.println(Arrays.toString(solution.maxSlidingWindow(new int[]{1,3,-1,-3,5,3,6,7}, 3)));
        System.out.println(Arrays.toString(solution.maxSlidingWindow(new int[]{7,2,4}, 2)));
        System.out.println(Arrays.toString(solution.maxSlidingWindow(new int[]{1, 3, -1, -3, 5, 3, 6, 7}, 4)));
        System.out.println(Arrays.toString(solution.maxSlidingWindow(new int[]{1, -1}, 1)));
    }
}
```
