class Solution {
    // [0, n-1] n-k+1 k <= [1, n-1] n-k+1 k-1 + [0, n-k+1] n-k+1 1
    // [1, n-1] n-k+1 k-1 <= [2, n-1] n-k+1 k-2 + [1, n-k+1] n-k+1 1
    // 3 -1 5 5 6 7
    // 1 3 -1 -3 5 3 6 7
    public int[] maxSlidingWindow(int[] nums, int k) {
        if (k == 1 || nums.length == 0) return nums;
        int ansLen = nums.length-k+1, max = Integer.MIN_VALUE;
        int[] ans = new int[ansLen];
        for (int i = 0; i < k; i++) max = Math.max(nums[i], max);
        ans[0] = max;
        for (int i = 1; i < ansLen; i++) {
            if (nums[i-1] != ans[i-1]) {
                ans[i] = Math.max(nums[i+ansLen-2], ans[i-1]);
            } else  {
                max = Integer.MIN_VALUE;
                for (int j = 0; j < ansLen; j++) max = Math.max(nums[i+j], max);
                ans[i] = max;
            }
        }
        // for (int i = k-1; i < nums.length; i++) ans[i-k+1] = nums[i];
        // while (k-->1) for (int i = 0; i < ansLen; i++) ans[i] = Math.max(nums[k-1+i], ans[i]);
        return ans;
    }

    static void main() {
        Solution solution = new Solution();
        System.out.println(Arrays.toString(solution.maxSlidingWindow(new int[]{7,2,4}, 2)));
        System.out.println(Arrays.toString(solution.maxSlidingWindow(new int[]{1, 3, -1, -3, 5, 3, 6, 7}, 4)));
        System.out.println(Arrays.toString(solution.maxSlidingWindow(new int[]{1, -1}, 1)));
    }
}
