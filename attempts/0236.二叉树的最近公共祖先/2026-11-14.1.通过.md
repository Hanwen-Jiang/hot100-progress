## 代码

```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, x):
#         self.val = x
#         self.left = None
#         self.right = None

class Solution:
    def lowestCommonAncestor(self, root: 'TreeNode', p: 'TreeNode', q: 'TreeNode') -> 'TreeNode':
        def isMe(root):
            if root is None or root is p or root is q:
                return root
            left = isMe(root.left)
            right = isMe(root.right)
            if left and right:
                return root
            return left or right
        return isMe(root)
```
