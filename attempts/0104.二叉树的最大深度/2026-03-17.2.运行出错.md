```java
class Solution {
    public int maxDepth(TreeNode root) {
        if (root == null) return 0;
        TreeNode[] nodes = new TreeNode[]{root};
        int deep = 1;
        while (true) {
            TreeNode[] nodes_temp = new TreeNode[(int) Math.pow(2, deep)];
            int index = 0;
            for (int i = 0; i < Math.pow(2, deep-1); i++) {
                if (nodes[i] != null) {
                    TreeNode node = nodes[i];
                    if (node.left != null) nodes_temp[index++] = node.left;
                    if (node.right != null) nodes_temp[index++] = node.right;
                } else if (i == 0) {
                    return deep-1;
                }
            }
            nodes = nodes_temp;
            deep++;
        }
    }
}
```
