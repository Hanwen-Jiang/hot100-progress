## 代码

```java
class Solution {
    class Node {
        Node next;
        TreeNode val;
        public Node() {}
        public Node(TreeNode val) {this.val = val;}
        public Node(TreeNode val, Node node) {this.val = val;this.next = node;}
    }

    public int maxDepth(TreeNode root) {
        if (root == null) return 0;
        Node head = new Node(root);
        int deep = 0;
        while (true) {
            Node head_temp = new Node(), p = head_temp;
            if (head == null) return deep;
            while (head != null) {
                TreeNode node = head.val;
                if (node.left != null) {p.next = new Node(node.left);p = p.next;}
                if (node.right != null) {p.next = new Node(node.right);p = p.next;}
                head = head.next;
            }
            head = head_temp.next;
            deep++;
        }
    }
}
```
