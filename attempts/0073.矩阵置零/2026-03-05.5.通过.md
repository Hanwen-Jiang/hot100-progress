```java
class Solution {
    public void setZeroes(int[][] matrix) {
        if (matrix.length == 0) return;
        int ilen = matrix.length, jlen = matrix[0].length, index = 0, jndex = 0;
        boolean[] ilist = new boolean[ilen], jlist = new boolean[jlen];
        for (int i=0; i<ilen; i++) for (int j=0; j<jlen; j++) if (matrix[i][j] == 0) {ilist[i] = true; jlist[j] = true;}
        for (int i=0; i<ilen; i++) for (int j=0; j<jlen; j++) if (ilist[i] || jlist[j]) matrix[i][j] = 0;
    }
}
```
