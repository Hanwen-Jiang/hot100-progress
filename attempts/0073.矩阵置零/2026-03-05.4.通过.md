## 代码

```java
class Solution {
    public void setZeroes(int[][] matrix) {
        int m = matrix.length;
        if (m == 0) return;
        int n = matrix[0].length;
        boolean[] rows = new boolean[m];
        boolean[] cols = new boolean[n];
        for (int i = 0; i < m; i++) {
            int[] row = matrix[i];
            for (int j = 0; j < n; j++) {
                if (row[j] == 0) {
                    rows[i] = true;
                    cols[j] = true;
                }
            }
        }
        for (int i = 0; i < m; i++) {
            int[] row = matrix[i];
            if (rows[i]) for (int j = 0; j < n; j++) row[j] = 0;
            else for (int j = 0; j < n; j++) if (cols[j]) row[j] = 0;
        }
    }
}
```
