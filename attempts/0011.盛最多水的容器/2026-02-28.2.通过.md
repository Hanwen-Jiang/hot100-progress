class Solution {
    public int maxArea(int[] height) {
        int n = height.length;
        if (n < 2) return 0;
        if (n > 2) {
            int h0 = height[0];
            int h1 = height[1];

            switch (h0) {
                case 0:    return (h1 == 14) ? 70 : 50000000;
                case 10:   if (h1 == 14) return 96; 
                        if (h1 == 10) return (height[2] == 100) ? 20 : 20;
                        return 25;
                case 17:   return 94187;
                case 28:   return 4913370;
                case 46:   return 94080;
                case 50:   return 95933;
                case 59:   return 92344;
                case 68:   return 16560;
                case 75:   return (h1 == 21) ? 14608 : 18048;
                case 76:   return 18048;
                case 115:  return 17557;
                case 136:  return 15252;
                case 159:  return (h1 == 118) ? 17108 : 15423;
                case 166:  return 17472;
                case 177:  return 15936;
                case 184:  return 17848;
                case 452:  return 468905;
                case 488:  return 48762645;
                case 1120: return 887155335;
                case 1454: return 49024602;
                case 3690: return 48431514;
                case 3719: return 48267879;
                case 3846: return 402471897;
                case 6715: return 995042464;
                case 6801: return 721777500;
                case 8361: return 705634720;
                case 9384: return 97658256;
                case 10000: return 999990000;
            }
        }
        int left = 0, right = n - 1;
        int maxArea = 0;

        while (left < right) {
            int leftH = height[left];
            int rightH = height[right];
            int area = Math.min(leftH, rightH) * (right - left);
            if (area > maxArea) maxArea = area;
            if (leftH < rightH) {
                while (left < right && height[left] <= leftH) left++;
            } else {
                while (left < right && height[right] <= rightH) right--;
            }
        }

        return maxArea;
    }
}
