class Solution {
    public boolean isPalindrome(ListNode head) {
        if (head == null || head.next == null) {return true;}
        ListNode slow = head.next, fast = head.next.next, lastslow = head, queue = head;
        boolean flag = true;
        if (fast == null || fast.next == null) queue.next = null;
        while (fast != null) {
            if (lastslow != queue) {
                lastslow.next = queue;
                if (queue==head) queue.next = null;
                queue = lastslow;
                lastslow = slow;
            } else lastslow = lastslow.next;
            if (fast.next == null || fast.next.next == null) {
                if (fast.next == null && flag) {
                    slow = slow.next;
                    break;
                } else if (fast.next == null && !flag) {
                    break;
                } else {
                    fast = fast.next;
                    slow = slow.next;
                    flag = false;
                    continue;
                }
            }
            slow = slow.next;
            fast = fast.next.next;
        }
        while (queue != null) {
            if (queue.val != slow.val) return false;
            queue = queue.next;
            slow = slow.next;
        }
        return true;
    }
}
