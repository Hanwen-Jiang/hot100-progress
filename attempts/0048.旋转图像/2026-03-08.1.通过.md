```java
class Solution {
    public void rotate(int[][] matrix) {
        // cur1 - x cur2 - y cur3 - x cur4 - y
        int n = matrix.length, cur1 = 0, cur2 = n-1;
        while (cur1 < cur2) {
            for (int i = 0; i < cur2-cur1; i++) {
                int temp = matrix[cur1][cur1+i];
                // matrix[cur1][i] = 1
                // matrix[i][cur2] = 3
                // matrix[cur2-i][cur1] = 7
                // matrix[cur2][cur2-i] = 9

                // matrix[cur1][i] = 2
                // matrix[i][cur2] = 6
                // matrix[cur2-i][cur1] = 4
                // matrix[cur2][cur2-i] = 8
                matrix[cur1][cur1+i] = matrix[cur2-i][cur1];
                matrix[cur2-i][cur1] = matrix[cur2][cur2-i];
                matrix[cur2][cur2-i] = matrix[cur1+i][cur2];
                matrix[cur1+i][cur2] = temp;
            }
            cur1++;cur2--;
        }
        return;
    }
}
```
