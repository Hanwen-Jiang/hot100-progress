## 代码

```java
class Solution {
    public void rotate(int[] nums, int k) {
        // [1,2,3,4,5,6,7]
        // [4,3,2,1,7,6,5]
        // [5,6,7,1,2,3,4]
        int len = nums.length, index = 0;
        k=k%len;
        if (len == 0 || len == 1 || k == 0) return;
        int[] res = new int[len];
        for (int i=len-k; i<len; i++) res[index++] = nums[i];
        for (int i=0; i<len-k; i++) res[index++] = nums[i];
        for (int i=0; i<len; i++) nums[i]=res[i];
    }
}
```
