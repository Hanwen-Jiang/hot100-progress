```python
class Solution:
    def minDistance(self, word1: str, word2: str) -> int:
        if word1 == word2:
            return 0
        prev = [0] * len(word2)
        for c1 in word1:
            cur = [0] * len(word2)
            for j, c2 in enumerate(word2):
                x1, x2, x3 = (cur[j-1], prev[j-1], prev[j]) if j>0 else (inf, inf, prev[j])
                cur[j] = min(x1, x2, x3)+1 if c1!=c2 else x2
            print(cur)
            prev = cur
        return prev[-1]
```
