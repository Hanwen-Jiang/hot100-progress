class Solution {
    public static void reverse(int[] arr) {
        int left = 0;
        int right = arr.length - 1;
        while (left < right) {
            int temp = arr[left];
            arr[left] = arr[right];
            arr[right] = temp;
            left++;
            right--;
        }
    }
    public int trap(int[] height) {
        int area = 0, left = 0, right, interArea = 0, lh, area1 = 0;
        for (int i = 0; i < 2; i++) {
            if (i == 1) {
                reverse(height);
                area1 = area;
                area = 0;
                left = 0;
            }
            while(left < height.length-1) {
                if (height[left] == 0) {
                    left++;
                    continue;
                }
                lh = height[left];
                right = left + 1;
                interArea = 0;
                while(right < height.length && height[right] < lh) {
                    interArea += height[right];
                    right++;
                }
                if (right >= height.length || height[right] < lh) {
                    left++;
                    continue;
                }
                area += lh*(right - left - 1) - interArea;
                left = right;
            }
        }
        return Math.max(area, area1);
    }

    static void main() {
        Solution solution = new Solution();
        System.out.println(solution.trap(new int[] {9,8,2,6}));
        System.out.println(solution.trap(new int[] {4,2,3}));
        System.out.println(solution.trap(new int[] {0,1,0,2,1,0,1,3,2,1,2,1}));
        System.out.println(solution.trap(new int[] {4,2,0,3,2,5}));
    }
}
