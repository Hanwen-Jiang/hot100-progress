class Solution {
    ListNode reverseK(ListNode head, int k) {
        ListNode dummy = new ListNode(0, head);
        int count = k;
        while (k-- > 1) {
            if (head.next == null) {
                k=count-k;
                return reverseK(dummy.next, k);
            }
            ListNode cur = head.next, h = dummy.next;
            dummy.next = cur;
            head.next = cur.next;
            cur.next = h;
        }
        return dummy.next;
    }
    public ListNode reverseKGroup(ListNode head, int k) {
        ListNode dummy = new ListNode(0, head), p = head, tail = dummy;
        while (p != null) {
            int kc = k;
            p = reverseK(p, k);
            ListNode t = tail.next;
            if (t == p) return dummy.next;
            tail.next = p;
            tail = t;
            while (kc-- > 0) p = p.next;
        }
        return dummy.next;
    }
}
