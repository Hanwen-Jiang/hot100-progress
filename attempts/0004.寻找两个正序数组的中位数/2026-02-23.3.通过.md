```java
class Solution {
    public double findMedianSortedArrays(int[] nums1, int[] nums2) {
        
        int l = nums1.length + nums2.length;
        int i1 = 0, i2 = 0;
        for(int i=0; i<l/2; i++) {
            if(i1 < nums1.length && (i2 >= nums2.length || nums1[i1] <= nums2[i2])) i1++;
            else i2++;
        }
        int val1Right = (i1 < nums1.length) ? nums1[i1] : Integer.MAX_VALUE;
        int val2Right = (i2 < nums2.length) ? nums2[i2] : Integer.MAX_VALUE;
        double k = val1Right>val2Right?val2Right:val1Right;
        if (l%2==1) return k;
        int val1Left = (i1 > 0) ? nums1[i1 - 1] : Integer.MIN_VALUE;
        int val2Left = (i2 > 0) ? nums2[i2 - 1] : Integer.MIN_VALUE;
        double k0 = val1Left>val2Left?val1Left:val2Left;
        return (k0+k)/2;
    }
}
// 0, 1
// 0, 0
// 1, 1
```
