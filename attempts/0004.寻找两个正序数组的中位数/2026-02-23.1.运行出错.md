class Solution {
    public double findMedianSortedArrays(int[] nums1, int[] nums2) {
        int l = nums1.length + nums2.length, i1 = 0, i2 = 0, t1 = nums1[0]<nums2[0] ? 1 : 0;
        double k, k0;
        for(int i=0; i<l/2; i++) {
            if(i1 >= nums1.length-1 || nums1[i1] > nums2[i2]) {
                if (t1==0) {t1=1; continue;}
                else i2++;
            }
            if(i2 >= nums2.length-1 || nums1[i1] <= nums2[i2]) {
                if (t1==1) {t1=0; continue;}
                else i1++;
            }
        }
        k = t1==1? nums1[i1]: nums2[i2];
        if (l%2==1) return k;
        int temp2 = i2==0? Integer.MIN_VALUE : nums2[i2-1], temp1 = i1==0? Integer.MIN_VALUE : nums1[i1-1];
        if (t1==0) k0 = temp2>nums1[i1]? temp2:nums1[i1];
        else k0 = temp1>nums2[i2]? temp1:nums2[i2];
        return (k0+k)/2;
    }
}

// 1, 1
