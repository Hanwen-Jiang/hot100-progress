```python
class Solution:
    def permute(self, nums: list[int]) -> list[list[int]]:
        used, res, path = [False]*len(nums), [], []
        def dfs():
            if len(nums) == len(path):
                res.append(path)
            for i, x in enumerate(nums):
                if used[i]:
                    continue
                used[i]=True
                path.append(x)
                dfs()
                path.pop()
                used[i]=False
        dfs()
        return res
```
