## 代码

```python
class Solution:
    def permute(self, nums: List[int]) -> List[List[int]]:
        res, path, used = [], [], [False] * len(nums)
        def dfs():
            if len(path) == len(nums):
                res.append(path[:])
            for i, x in enumerate(nums):
                if used[i]:
                    continue
                used[i] = True
                path.append(nums[i])
                dfs()
                path.pop()
                used[i] = False
        dfs()
        return res
```
