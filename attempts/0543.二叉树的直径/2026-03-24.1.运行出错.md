```java
class Solution {
    int depth(TreeNode root) {
        if (root == null) return 0;
        return max(depth(root.left), depth(root.right)) + 1;
    }
    public int diameterOfBinaryTree(TreeNode root) {
        return depth(root.left) + depth(root.right);
    }
}
```
